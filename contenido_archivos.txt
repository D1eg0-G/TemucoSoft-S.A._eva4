// src/admin-cliente/config/AuthContext.jsx

import { createContext, useState, useEffect } from "react";
import axios from "axios";

export const AuthContext = createContext();

const PLAN_CONFIG = {
  basico: {
    name: "Plan Básico",
    apiUrl: import.meta.env.VITE_API_BASICO_URL,
    modules: ["dashboard", "products", "inventory", "sale", "cashregister"],
  },
  estandar: {
    name: "Plan Estándar",
    apiUrl: import.meta.env.VITE_API_ESTANDAR_URL,
    modules: [
      "dashboard",
      "products",
      "inventory",
      "sale",
      "cashregister",
      "branches",
      "providers",
      "purchases",
      "orders",
      "reports",
    ],
  },
  premium: {
    name: "Plan Premium",
    apiUrl: import.meta.env.VITE_API_PREMIUM_URL,
    modules: [
      "dashboard",
      "products",
      "inventory",
      "sale",
      "cashregister",
      "branches",
      "providers",
      "purchases",
      "orders",
      "reports",
      "gestion-user",
      "subscription",
    ],
  },
};

const MASTER_API =
  import.meta.env.VITE_API_MASTER_URL || "http://localhost:8000";

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [plan, setPlan] = useState(null);
  const [apiUrl, setApiUrl] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    // Debug: log values read from localStorage to detect bad URLs
    const storedUser = localStorage.getItem("user");
    const storedPlan = localStorage.getItem("plan");
    const storedToken = localStorage.getItem("access_token");
    try {
      console.debug("AuthContext: storedUser:", storedUser);
      console.debug("AuthContext: storedPlan:", storedPlan);
      console.debug(
        "AuthContext: storedToken:",
        storedToken ? "<token>" : null
      );
    } catch (e) {
      // ignore
    }

    if (storedUser && storedPlan && storedToken) {
      const userData = JSON.parse(storedUser);
      const planData = JSON.parse(storedPlan);

      setUser(userData);
      setPlan(planData);
      const candidateApi = PLAN_CONFIG[planData.tipo]?.apiUrl;
      // Validate apiUrl looks like an http(s) URL
      if (candidateApi && /^https?:\/\//.test(candidateApi)) {
        setApiUrl(candidateApi);
      } else {
        console.warn(
          "AuthContext: apiUrl from plan is invalid, using null:",
          candidateApi
        );
        setApiUrl(null);
      }
    }
    setLoading(false);
  }, []);

  const login = async (email, password) => {
    try {
      // 1. Obtener routing desde TemucoSoft Master
      console.debug("AuthContext.login: MASTER_API ->", MASTER_API);
      const routingRes = await axios.post(
        `${MASTER_API}/api/empresas/login-router/`,
        {
          email,
          password,
        }
      );

      const {
        empresa_id,
        plan: planTipo,
        instance_url,
        modulos,
      } = routingRes.data;

      // 2. Autenticar en la instancia correcta
      // Validate instance_url before using it
      if (!instance_url || !/^https?:\/\//.test(instance_url)) {
        throw new Error("instance_url inválida desde routing: " + instance_url);
      }

      const tokenRes = await axios.post(`${instance_url}/api/token/`, {
        username: email,
        password,
      });

      const { access, refresh } = tokenRes.data;

      // 3. Obtener datos del usuario
      const userRes = await axios.get(`${instance_url}/api/usuarios/me/`, {
        headers: { Authorization: `Bearer ${access}` },
      });

      const userData = { ...userRes.data, empresa_id };
      const planData = {
        tipo: planTipo,
        modulos,
        config: PLAN_CONFIG[planTipo],
      };

      setUser(userData);
      setPlan(planData);
      setApiUrl(instance_url);

      localStorage.setItem("user", JSON.stringify(userData));
      localStorage.setItem("plan", JSON.stringify(planData));
      localStorage.setItem("access_token", access);
      localStorage.setItem("refresh_token", refresh);

      return { success: true };
    } catch (error) {
      console.error("Login error:", error);
      return {
        success: false,
        error: error.response?.data?.error || "Error de autenticación",
      };
    }
  };

  const logout = () => {
    setUser(null);
    setPlan(null);
    setApiUrl(null);
    localStorage.clear();
  };

  const hasModule = (moduleName) => {
    return plan?.modulos?.includes(moduleName) || false;
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        plan,
        apiUrl,
        loading,
        login,
        logout,
        hasModule,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

// src/core/layout/Sidebar/Sidebar.jsx

import React, { useState, useContext } from "react";
import { useNavigate, useLocation } from "react-router-dom";
import { AuthContext } from "../../../admin-cliente/config/AuthContext";
import "./Sidebar.css";
import logo from "/src/assets/Logo_v.png";
import {
  Menu,
  Settings,
  LogOut,
  CreditCard,
  ChevronUp,
  ChevronDown,
  Crown,
} from "lucide-react";

const Sidebar = ({
  menuItems,
  basePath = "/cliente",
  companyName = "Mi Empresa",
}) => {
  const { user, plan, hasModule, logout } = useContext(AuthContext);

  const [isCollapsed, setIsCollapsed] = useState(false);
  const [isSettingsOpen, setIsSettingsOpen] = useState(false);

  const navigate = useNavigate();
  const location = useLocation();

  const handleNavigation = (path) => {
    const root = basePath.startsWith("/") ? basePath : `/${basePath}`;
    const finalBase = root.endsWith("/") ? root.slice(0, -1) : root;
    const cleanPath = path.startsWith("/") ? path.slice(1) : path;
    navigate(`${finalBase}/${cleanPath}`);
  };

  const isActive = (path) => {
    const root = basePath.startsWith("/") ? basePath : `/${basePath}`;
    return location.pathname.includes(`${root}/${path}`);
  };

  const handleLogout = (e) => {
    e.preventDefault();
    logout();
    navigate("/");
  };

  // Filtrar menú por ROL y PLAN
  const filteredMenuItems = menuItems.filter((item) => {
    // Verificar rol
    if (item.allowedRoles && !item.allowedRoles.includes(user?.role)) {
      return false;
    }

    // Verificar módulo del plan
    if (item.module && !hasModule(item.module)) {
      return false;
    }

    return true;
  });

  const userRole = user?.role || "cargando...";
  const planName = plan?.config?.name || "Cargando...";

  return (
    <aside className={`sidebar-container ${isCollapsed ? "collapsed" : ""}`}>
      {/* Header */}
      <div className="sidebar-header">
        <div className={`brand-wrapper ${isCollapsed ? "hidden" : ""}`}>
          <img src={logo} alt="Logo" className="brand-logo" />
          <div style={{ display: "flex", flexDirection: "column", gap: "2px" }}>
            <span className="brand-name">{companyName}</span>
            <span
              style={{
                fontSize: "11px",
                color: "#64748b",
                fontWeight: "500",
              }}
            >
              {planName}
            </span>
          </div>
        </div>

        <button
          className="toggle-btn"
          onClick={() => setIsCollapsed(!isCollapsed)}
        >
          <Menu size={24} color="#0e3c66" />
        </button>
      </div>

      {/* Navigation */}
      <nav className="sidebar-nav">
        <ul>
          {filteredMenuItems.map((item, index) => (
            <li
              key={index}
              className={`nav-item ${isActive(item.path) ? "active" : ""}`}
            >
              <a
                href="#"
                className="nav-link"
                onClick={(e) => {
                  e.preventDefault();
                  handleNavigation(item.path);
                }}
                title={isCollapsed ? item.title : ""}
              >
                <span className="nav-icon">{item.icon}</span>
                {!isCollapsed && (
                  <>
                    <span className="nav-text">{item.title}</span>
                    {/* Badge para sucursales */}
                    {item.badge && plan?.config?.limits?.maxBranches && (
                      <span
                        style={{
                          fontSize: "10px",
                          backgroundColor: "#3b82f6",
                          color: "white",
                          padding: "2px 6px",
                          borderRadius: "4px",
                          marginLeft: "auto",
                        }}
                      >
                        {plan.config.limits.maxBranches === null
                          ? "∞"
                          : plan.config.limits.maxBranches}
                      </span>
                    )}
                  </>
                )}
              </a>
            </li>
          ))}
        </ul>
      </nav>

      {/* Upgrade Banner */}
      {!isCollapsed && plan?.tipo !== "premium" && (
        <div
          style={{
            margin: "12px",
            padding: "12px",
            background: "linear-gradient(135deg, #3b82f6 0%, #8b5cf6 100%)",
            borderRadius: "8px",
            color: "white",
          }}
        >
          <div
            style={{
              display: "flex",
              alignItems: "center",
              gap: "8px",
              marginBottom: "8px",
            }}
          >
            <Crown size={16} />
            <strong style={{ fontSize: "13px" }}>Upgrade a Premium</strong>
          </div>
          <p
            style={{
              fontSize: "11px",
              margin: "0 0 10px 0",
              opacity: 0.9,
            }}
          >
            Desbloquea todas las funciones
          </p>
          <button
            onClick={() => handleNavigation("Subcription")}
            style={{
              width: "100%",
              padding: "8px",
              backgroundColor: "white",
              color: "#3b82f6",
              border: "none",
              borderRadius: "6px",
              fontSize: "12px",
              fontWeight: "600",
              cursor: "pointer",
            }}
          >
            Ver Planes
          </button>
        </div>
      )}

      {/* Footer */}
      <div className="sidebar-footer">
        {/* Ocultar Mi cuenta si es super-admin */}
        {userRole !== "super-admin" && (
          <div className="settings-group">
            <a
              href="#"
              className="nav-link footer-link"
              onClick={(e) => {
                e.preventDefault();
                if (!isCollapsed) setIsSettingsOpen(!isSettingsOpen);
              }}
            >
              <span className="nav-icon">
                <Settings size={20} />
              </span>
              {!isCollapsed && (
                <>
                  <span className="nav-text" style={{ flex: 1 }}>
                    Mi cuenta
                  </span>
                  {isSettingsOpen ? (
                    <ChevronUp size={16} />
                  ) : (
                    <ChevronDown size={16} />
                  )}
                </>
              )}
            </a>

            {/* Submenu de Suscripción */}
            {!isCollapsed && isSettingsOpen && hasModule("subscription") && (
              <div className="settings-submenu">
                <a
                  href="#"
                  className={`nav-link submenu-link ${
                    isActive("Subcription") ? "active" : ""
                  }`}
                  onClick={(e) => {
                    e.preventDefault();
                    handleNavigation("Subcription");
                  }}
                >
                  <span className="nav-icon">
                    <CreditCard size={18} />
                  </span>
                  <span className="nav-text">Mi Suscripción</span>
                </a>
              </div>
            )}
          </div>
        )}

        {/* Logout */}

        <a
          href="#"
          className="nav-link footer-link logout"
          onClick={handleLogout}
        >
          <span className="nav-icon">
            <LogOut size={20} />
          </span>
          {!isCollapsed && <span className="nav-text">Salir</span>}
        </a>
      </div>
    </aside>
  );
};

export default Sidebar;
